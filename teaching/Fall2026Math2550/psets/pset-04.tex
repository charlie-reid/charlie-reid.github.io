\begin{exercise}[10 points; part of Rudin 2.9]
    Let $E$ be a subset of a metric space. Recall that the \emph{interior} of $E$, denoted $E^\circ$, is the set of all interior points of $E$.
    \begin{enumerate}
        \item Prove that $E^\circ$ is always open.
        \item Prove that $E$ is open if and only if $E^\circ = E$.
        \item Prove that, if $G$ is an open subset of $E$, then $G \subset E^\circ$.
    \end{enumerate}
\end{exercise}

\end{exercise}


\begin{exercise}[10 points; Rudin 2.25, modified] A metric space which has a dense subset which is at most countable is called \textit{separable}. For example, $\Q$ is a countable, dense subset of $\R$, so $\R$ is separable.
\begin{enumerate}
    \item 
Suppose that $K$ is a compact metric space. Prove that $K$ is separable.  (Hint:
first show that for every $n \in {\mathbb N}$, there are finitely many neighborhoods of radius $1/n$ whose union covers $K$.)

\item Give an example of a metric space which is not separable.
\end{enumerate}
\end{exercise}



\begin{exercise}[10 points; Rudin 2.19, modified]
We say a metric space $X$ is \textit{connected} if it cannot be written as a disjoint union of two nonempty open sets. (This definition is slightly different from that in Rudin; you can show it is equivalent. )

\begin{enumerate}
\item Show that $X$ is connected if and only if the only clopen (closed and open) subsets of $X$ are $\varnothing$ and $X$. 

\item Fix $p \in X$ and $\delta > 0$.  Define $A = \{q \in X \mid d(p,q) < \delta \}$.
Define $B = \{q \in X \mid d(p,q) > \delta \}$.  Show that $A$ and $B$ are open sets.

\item Prove that every connected metric space with at least two points is uncountable.  (Hint: use (2))


\end{enumerate}

\end{exercise}
