\begin{exercise}[8 points]  \ 
\begin{enumerate}
 \item    Prove that the interval $(0,1)$ in $\R$ is not complete.
 \item    Suppose $X$ is a set equipped with the discrete metric. Is $X$ complete?
\end{enumerate} 
\end{exercise}




\begin{exercise}[8 points; Rudin 3.24]

Suppose $(p_n)$ and $(q_n)$ are Cauchy sequences in a metric space $X$. Prove 
that the sequence $(d(p_n,q_n))$ in $\R$
has a limit.

\end{exercise}


\begin{exercise} [8 points] Let $(s_n)$ be the sequence defined by
\[s_1 = 0 \;\;\;\; s_{2n} = s_{2n-1} + 1 \;\;\;\; s_{2n+1} = \frac{s_{2n} }{2}\]
Find the upper and lower limits of $(s_n)$ (a.k.a $\limsup$ and $\liminf$.)  Prove your answers.
\end{exercise}

\begin{exercise}[8 points; Rudin 3.21]
Recall that for a subset $E$ of a metric space, we
define ${\mathrm{diam}}\, E = \sup \{ d(p,q) \mid p,q \in E \}$.  Suppose $(E_n)$ is a sequence of closed, nonempty sets in a complete metric space $X$,
with $E_n \supset E_{n+1}$, and $\lim_{n \to \infty} {\mathrm{diam}}\,E_n = 0$.  Prove that 
$\cap_{n=1}^\infty E_n$ consists of exactly one point.
\end{exercise}

\noindent\hrulefill



\begin{exercise}[not for credit]
Suppose $E \subset X$. We say $E$ is \ti{sequentially closed} if every sequence $(p_n)$ in $E$ which is convergent in $X$
is actually convergent in $E$. Prove that $E$ is closed if and only if it is sequentially closed.
\end{exercise}

\begin{exercise}[not for credit]
Suppose $(p_{2n})$ and $(p_{2n+1})$ both converge to the same limit $q$. 
Prove that $(p_n)$ converges to $q$. 
\end{exercise}



\begin{exercise}[not for credit; challenge] This exercise explains the name of $\limsup$. Show that for any sequence of real numbers $(s_n)$, 
\[ \limsup_{n\to\infty} s_n = \lim_{n\to\infty} \sup\{s_m | m\geq n\}\]
\end{exercise}

\begin{exercise}[not for credit; challenge]
Let $r$ be a an irrational positive real number. Let \[s_n = nr - \lfloor nr \rfloor\] where $\lfloor a \rfloor$ denotes the greatest integer less than the real number $a$. Show that every point in the interval $[0,1]$ is a subsequential limit of $(s_n)$. (This is a bit involved, at least the way I did it. In dynamics, one would phrase roughly this result as "an irrational rotation generates a minimal action on the circle.")
\end{exercise}
