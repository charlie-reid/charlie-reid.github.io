\begin{exercise}[3 points]
Let $F$ be any ordered field.
Suppose $A \subset F$ has a supremum. Suppose $\eps \in F$
with $\eps > 0$. 
Prove that there exists some $x \in A$ such that $x > (\sup A) - \eps$.
\end{exercise}


\begin{exercise}[5 points; Rudin 1.5]
Let $A \subset \R$ be nonempty and bounded below. Define
\begin{equation}
-A = \{-x\ \mid x \in A\} \, .  
\end{equation}
Prove that $\inf A = - \sup(-A)$.
\end{exercise}


\begin{exercise}[10 points]
For each of the following, $A$ is a subset of $\mathbb{Q}$.
Answer the following in each case, and prove your answers: 
\begin{itemize}
\item Is $A$ bounded above? 
\item Does $A$ have a maximum element, and if so, what is it? 
\item Does $A$ have a supremum in $\Q$, and if so, what is it?
\end{itemize}
\begin{enumerate}
  \item $A = \{ -\frac{1}{n} \mid n \in \bbN \}$.
  \item $A = \{ x \in \Q \mid 0 < x < 1 \}$.
  \item $A = \{ x \in \Q \mid 0 < x < t \}$ where $t$ is a positive irrational number.
\end{enumerate}

\end{exercise}


\begin{exercise}[5 points] This exercise is a reminder that the cardinality of sets is sometimes a counterintuitive notion of size.

\begin{enumerate}
 
 \item
Suppose $a, b \in \R$, with $a < b$.
Prove that there is a bijection from $(a,b)$ to $(0,1)$.

\item Prove that there is a bijection from $(0,1)$ to $(1,\infty)$.

\item Suppose $a,b \in \R$. 
Prove that there is a bijection from $(a,\infty)$ to $(b,\infty)$.
\end{enumerate}
\end{exercise}


\begin{exercise}[10 points]
Decide if the following sets are countable or uncountable. Prove your answers.
\begin{enumerate}
\item The set of irrational numbers.
\item The set of sequences of natural numbers $n_1,n_2,n_3,\dots$.
\item The set of sequences of natural numbers $n_1,n_2,n_3,\dots$ such that there exists some $l\in \mathbb{N}$ such that $n_k=0$ for all $k>l$. 
\end{enumerate}
\end{exercise}

\begin{exercise}[optional, not for credit; Rudin 1.8 and 1.9]

Define the complex numbers as the set 
\begin{equation}
\C= \{a+b \I \mid a,b \in \R \}
\end{equation}
with the addition and multiplication rules given by
\begin{itemize}
\item $(a+b\I)+(a'+b'\I) = (a+a') + (b+b')\I$,
\item $(a+b\I)(a'+b'\I) = (aa'-bb') + (ab'+a'b)\I$.
\end{itemize}
\begin{enumerate}
\item Prove that $\C$ with these operations (and an appropriate $0$, $1$, negation, inversion) is a field.
\item Prove that $\C$ with these operations cannot be made into an ordered field.
\item Define the \emph{lexicographic order} on $\C$ by defining 
\begin{equation}
a+b\I<c+d\I \quad \iff \quad a<c \ \text{ or } \ (a=c \text{ and } b<d). 
\end{equation}
Prove that this makes $\C$ into an ordered \emph{set} (though not an ordered \emph{field}).
  
\item Does the lexicographic order on $\C$ have the least upper bound property?
\end{enumerate}

\end{exercise}
