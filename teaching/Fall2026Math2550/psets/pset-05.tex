\begin{exercise}[8 points]
	Use the definition of convergence to show that $\displaystyle{\lim_{n\to\infty}\frac{2n+1}{3n-1}=\frac{2}{3}}$.
\end{exercise}


\begin{exercise}[8 points]
	Suppose $(p_n)$ is a sequence in a metric space $X$ that converges to some $q \in X$. Show that the set $\{p_n\mid n\in\bbN\} \cup \{q\}$ is compact.
\end{exercise}


\begin{exercise}[10 points; Squeeze Theorem] 
    Let $(a_n)$, $(s_n)$ and $(b_n)$ be sequences of real numbers such that $a_n \leq s_n \leq b_n$ for all $n\in \bbN$. Prove that if $x_n\to t$ and $y_n\to t$, then $s_n\to t$. 
\end{exercise}

\begin{exercise}[10 points]
Consider two possible norms on $\R^2$: for a vector $v = (v_1, v_2)$,
$$\norm{v}_2 = \sqrt{v_1^2 + v_2^2} \quad , \qquad \norm{v}_1 = \abs{v_1} + \abs{v_2} \, .$$
Then let 
$$d_2(x,y) = \norm{x-y}_2 \quad , \qquad  d_1(x,y) = \norm{x-y}_1 \, .$$
Either $d_1$ or $d_2$ makes $\R^2$ into a metric space (you need not prove this).

Now, suppose $(x_n)$ is a sequence in $\R^2$.
Prove that $x_n \to x$ in the metric $d_2$ if and only if $x_n \to x$ in the metric $d_1$.
(Hint: first prove some inequalities relating $d_1$ to $d_2$.)
\end{exercise}

\begin{exercise}[not for credit; extra practice]
Suppose $E \subset X$. We say $E$ is \ti{sequentially closed} if every sequence $(p_n)$ in $E$ which is convergent in $X$
is actually convergent in $E$. Prove that $E$ is closed if and only if it is sequentially closed.
\end{exercise}

\begin{exercise}[not for credit; extra practice] (This is a modification of Rudin's exercise 2.26) We say a metric space $X$ is \textit{sequentially compact} if every sequence in $X$ has a convergent subsequence. In class we will show that compactness implies sequential compactness. Here we show the converse.
\begin{enumerate}
\item  A metric space is \textit{separable} if it admits a countable dense subset. Show that if $X$ is sequentially compact, it is separable. 

\item A \textit{base} for a metric space $X$ is a family of open sets $\mathcal{U} = \{U_\alpha\}$ such that for every open set $V$, and every point $x\in V$, there is an element of the family $U_\alpha$ such that $x\in U_\alpha \subset V$. Equivalently, every open set of $X$ can be expressed as a union of a subfamily of $\mathcal{U}$. Show that if $X$ is separable, it admits a countable base. 

\item Show that if $X$ admits a countable base, than every open cover of $X$ has a countable subcover.

\item Show that if $F_1 \supset F_2 \supset \cdots$ is a nested sequence of nonempty closed subsets of a sequentially compact metric space $X$,  then $\bigcap_{k\in \bbN} F_k$ is nonempty.

\item Show that every countable cover of a sequentially compact metric space has a finite subcover. Hint:  Suppose $\{G_i\}_{i\in \bbN}$ is a countable cover with no finite subcover, and consider the subsets $(G_1\cup \dots \cup G_k)^c$.

\end{enumerate}
\end{exercise}
