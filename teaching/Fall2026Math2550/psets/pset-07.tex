\[\sum_{n=2}^\infty \frac{1}{n(\log_2(n))^2}\]

\end{exercise}


\begin{exercise}[Rudin 3.9; 8 points] Find the radius of convergence of each of the following power series.
\begin{multicols}{4}
\begin{enumerate}[label=(\alph*)]
\item $\sum n^3 z^n$
\item $\sum \frac{2^n}{n!} z^n$
\item $\sum \frac{2^n}{n^2} z^n$
\item $\sum \frac{n^3}{3^n} z^n$
\end{enumerate}
\end{multicols}
\end{exercise}


\begin{exercise}[8 points] Let
$
a_{m,n} =
\begin{cases}
m & \text{if } m = n, \\
- m & \text{if } n = m + 1, \\
0 & \text{otherwise.}
\end{cases}
$
\begin{enumerate}[label=(\alph*)]
\item \[\text{Does the series}
\sum_{m=1}^{\infty} \left( \sum_{n=1}^{\infty} a_{m,n} \right)
\text{converge? If so, to what value?}\]
\item \[\text{Does the series}
\sum_{n=1}^{\infty} \left( \sum_{m=1}^{\infty} a_{m,n} \right)
\text{converge? If so, to what value?}\]

\end{enumerate}
\end{exercise}


\begin{exercise}[10 points]

Suppose that $(x_n)$ and $(y_n)$ are sequences in $\R$, with all $x_n > 0$ and $y_n > 0$, and for some $L \in 
(0,\infty)$
we have
$$
\lim_{n \to \infty} \frac{x_n}{y_n} = L \, .
$$
Prove that $\sum_{n=1}^\infty x_n$ converges if and only if 
$\sum_{n=1}^\infty y_n$ converges.

(This is sometimes called the ``limit-comparison test'' for convergence of series.)

\end{exercise}

\begin{exercise}[not for credit; Rudin 3.24, in part]
Let $X$ be a metric space.
$X$ might or might not be complete.
If $X$ is not complete, it would be nice
to know how to ``fill in the holes'' to
make it complete.
This exercise explains a way of doing so:
it constructs a new complete 
metric space $X^*$
which has $X$ as a subset. This construction is called \ti{completion}, and is one way to construct $\R$ from $\Q$.

We call two Cauchy sequences $(p_n)$,
$(q_n)$ in $X$ 
\ti{equivalent} if $d(p_n, q_n) \to 0$.
We write this relation as $(p_n) \sim (q_n)$.
\begin{enumerate}
\item Prove that this 
is an equivalence relation, i.e.
\begin{enumerate}
\item Any Cauchy sequence $(p_n)$ 
has $(p_n) \sim (p_n)$,
\item If $(p_n) \sim (q_n)$ then $(q_n) \sim (p_n)$,
\item If $(p_n) \sim (q_n)$
and $(q_n) \sim (r_n)$,
then $(p_n) \sim (r_n)$.
\end{enumerate}

\item If $(p_n) \sim (p'_n)$
and $(q_n) \sim (q'_n)$,
prove that
$$
\lim_{n \to \infty} d(p_n,q_n) = \lim_{n \to \infty} d(p'_n,q'_n) \, .
$$
(Note that the limit does exist, by the
result of an earlier exercise.)
\end{enumerate}

Now we divide the set of Cauchy sequences in $X$ into \ti{equivalence classes}: 
any two elements of a given class $P$ are
equivalent, and elements of different
classes $P$, $Q$ are not equivalent.
Let $X^*$ be the set of all equivalence classes of Cauchy sequences in $X$. 

Then, define a distance function $\Delta$
on $X^*$ as follows: if $(p_n)$ is in the class $P$, and $(q_n)$ is in the class $Q$,
then 
$$
\Delta(P,Q) = \lim_{n \to \infty} d(p_n,q_n) \, .
$$
The previous parts show that this distance function is well defined.
\begin{enumerate}[resume]
\item Prove that the distance function $\Delta$ makes $X^*$ into a metric space.
\item Prove that $X^*$ with this distance function is complete.
\item Consider the map $\phi: X \to X^*$
which maps any $x \in X$ to the class
of the Cauchy sequence $(x,x,x,\dots)$.
Prove that $\phi$ is injective and
$\Delta(\phi(x),\phi(y)) = d(x,y)$.
\end{enumerate}

\end{exercise}
