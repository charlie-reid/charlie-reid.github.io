\begin{exercise}[9 points; Rudin 3.6, modified]
Determine whether $\sum_{n=1}^\infty a_n$ converges or diverges if
\begin{enumerate}
\item $a_n = \sqrt{n+1} - \sqrt{n}$,
\item $a_n = \frac{\sqrt{n+1} - \sqrt{n}}{n}$,
\item $a_n = (n^{1/n} - 1)^n$.
\end{enumerate}

\end{exercise}


\begin{exercise}[6 points; Rudin 3.7]
Suppose that all $x_n \ge 0$ and
$\sum_{n=1}^\infty x_n$ converges.
Prove that
$\sum_{n=1}^\infty \frac{\sqrt{x_n}}{n}$
also converges. (Hint: use the Cauchy-Schwarz inequality.)
\end{exercise}

\begin{exercise}[10 points]

Consider the function $f: \R \to \R$ defined by
\begin{equation}
  f(x) = \begin{cases} 0 & \text{ if } x \notin \Q \, , \\
  x^2 & \text{ if } x \in \Q \, . \end{cases}
\end{equation}
Prove that:
\begin{enumerate}
\item  $\lim_{x \to 0} f(x) = 0$,
\item  $\lim_{x \to \alpha} f(x)$ does not exist if $\alpha \neq 0$.
\end{enumerate}

\end{exercise}

\begin{exercise}[6 points] It is interesting to think about which subsets of $\R$ can arise as the set of points of continuity of a function. For this problem, writing out proofs is optional; if you want, just give examples. 
\begin{enumerate}[label=(\alph*)] 
\item Give an example of a function $f:\R\to \R$ which is continuous at all points in $(-\infty, 0)$, and discontinous at all points in $[0,\infty)$.
\item Give an example of a function $f:\R\to \R$ which is continuous at all points in $(-\infty, 0]$, and discontinous at all points in $(0,\infty)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[not for credit; Rudin 3.16]
Fix a positive number $\alpha$. Choose $x_1 > \sqrt{\alpha}$, and define $x_2, x_3, x_4, \dots$
by the recursion formula
$$
x_{n+1} = \frac12 \left(x_n + \frac{\alpha}{x_n}\right) \, .
$$
\begin{enumerate}
    \item Prove that $(x_n)$ is monotonically decreasing and $\lim_{n \to \infty} x_n = \sqrt{\alpha}$.
    \item Define $\varepsilon_n = x_n - \sqrt{\alpha}$. Then show that
    $$
    \varepsilon_{n+1} = \frac{\varepsilon_n^2}{2 x_n} < \frac{\varepsilon_n^2}{2 \sqrt\alpha}
    $$
    and thus, setting $\beta = 2 \sqrt{\alpha}$
    $$
    \varepsilon_{n+1} < \beta \left(\frac{\varepsilon_1}{\beta}\right)^{2^n}
    $$
    \item This is a good algorithm for computing square roots, since the recursion formula is simple and the convergence is extremely rapid. For example, if $\alpha = 3$ and $x_1 = 2$,
    show that $\varepsilon_1 / \beta < \frac{1}{10}$ and therefore $\varepsilon_5 < 4 \cdot 10^{-16}$,
    $\varepsilon_6 < 4 \cdot 10^{-32}$.
\end{enumerate}

\end{exercise}
