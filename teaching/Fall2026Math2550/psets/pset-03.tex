\begin{exercise}[5 points]
Prove that $\bbN$ is not bounded above in $\R$.
\end{exercise}

\begin{exercise}[5 points]
    Let $X$ be any set, and let $d:X\times X\to \R$ be the discrete metric, defined by
    \[
    d(x,y)=\begin{cases}0&x=y \, \\ 1&x\neq y\end{cases}
    \]
    for all $x,y\in X$.
    \begin{enumerate}
        \item Prove that, with this distance function, $X$ is a metric space.
        \item For any $x\in X$, what is the neighbourhood $N_\epsilon(x)$ when $\epsilon = \frac{1}{2}$, $1$, and $2$?
        \item Which subsets of $X$ are open? Which are closed?
    \end{enumerate}
\end{exercise}


\begin{exercise}[5 points]
Show that the subset of $\R^2$ given by
$$E = \{(x,y) \in \R^2 \, \vert \, x < y\}$$ is open.
\end{exercise}

\begin{exercise}[10 points; Rudin 2.2+2.3, modified]

A real number $x$ is called \emph{algebraic} if there exist $n \in \bbN$ and $a_0, \dots, a_n \in \bbZ$, with $a_0 \neq 0$, such that
\begin{equation*}
 a_0 x^n + a_1 x^{n-1} + \cdots + a_{n-1} x + a_n = 0.
\end{equation*}

\begin{enumerate}
\item
Prove that $\sqrt{5}$ and $\sqrt{2 + \sqrt{3}}$ are algebraic.

\item 
Prove that the set of all algebraic real numbers is countable.  (You may
use without proof the fact that a polynomial of degree $n$ has at most
$n$ roots.)

\item Prove that there exist real numbers which are not algebraic.
\end{enumerate}
\end{exercise}

\begin{exercise}[10 points] In class, we will see that a sequence of bounded closed intervals $E_1,E_2, E_3,... $  with $E_{i+1}\subset E_i$ always has non-empty intersection. In this exercise we see that this is a very special property. 
\begin{enumerate}
\item Find a sequence of open intervals $U_1, U_2, U_3, ...$ in $\R$ with $U_{k+1} \subset U_k$ such that the intersection $\bigcap_{k\in \mathbb{N}} U_k$ is empty.

\item Find a sequence of closed sets $C_1, C_2, C_3, ...$ in $\R$ with $C_{k+1} \subset C_k$ such that the intersection $\bigcap_{k\in \mathbb{N}} C_k$ is empty.

\item Find a sequence of dense sets $D_1, D_2, D_3, ...$ in $\R$ with $D_{k+1} \subset D_k$ such that the intersection $\bigcap_{k\in \mathbb{N}} D_k$ is empty. Prove the $D_i$ you define are dense.
\end{enumerate}
\end{exercise}
